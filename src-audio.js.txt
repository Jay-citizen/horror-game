export class SoundEngine {
  constructor() {
    this.ctx = null;
    this.stepTimer = 0;
  }

  init() {
    if (!this.ctx && typeof window !== 'undefined') {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  playTone(freq, dur = 0.15, type = 'sine', vol = 0.12) {
    if (!this.ctx) return;
    const t = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, t);
    gain.gain.setValueAtTime(0.001, t);
    gain.gain.exponentialRampToValueAtTime(vol, t + 0.015);
    gain.gain.exponentialRampToValueAtTime(0.001, t + dur);
    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start(t);
    osc.stop(t + dur + 0.02);
  }

  paper() {
    this.playTone(620, 0.08, 'triangle', 0.08);
    this.playTone(840, 0.09, 'sine', 0.06);
  }

  clue() {
    this.playTone(440, 0.14, 'triangle', 0.12);
    setTimeout(() => this.playTone(587, 0.22, 'triangle', 0.12), 110);
  }

  step(surface = 'wood') {
    const base = surface === 'gravel' ? 145 : surface === 'concrete' ? 190 : 115;
    this.playTone(base + Math.random() * 25, 0.06, 'triangle', 0.05);
  }
}
